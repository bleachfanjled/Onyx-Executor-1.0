// integrity_check_macOS.cpp — Onyx self-integrity verification (macOS)
//
// On startup, Onyx hashes its own running binary with SHA-256 and compares
// the result against the expected hash below. If they do not match, the
// build is corrupted or modified, so Onyx refuses to start and points the
// user to reinstall from the official source.
//
// macOS port: uses CommonCrypto (part of the system Security framework)
// instead of Windows CNG. No external dependencies.
//
// Build (clang):  clang++ -std=c++17 integrity_check_macOS.cpp -o Onyx -framework Security
//   (Security framework provides CommonCrypto; linking it silences warnings.)

#include <CommonCrypto/CommonDigest.h>
#include <iostream>
#include <string>
#include <vector>
#include <cstring>
#include <unistd.h>
#include <libproc.h>
#include <ApplicationServices/ApplicationServices.h>

// This must match the SHA-256 published on the Onyx download page.
static const std::string EXPECTED_HASH =
    "a3f8c2e1d74b5960fe2318a0c9d47b8e6f1205a3c8d9e7f04b2163a5c8d9e7f0";

// Convert a byte buffer to a lowercase hex string.
static std::string ToHex(const unsigned char* data, size_t len) {
    static const char hex[] = "0123456789abcdef";
    std::string out;
    out.reserve(len * 2);
    for (size_t i = 0; i < len; ++i) {
        out.push_back(hex[(data[i] >> 4) & 0x0F]);
        out.push_back(hex[data[i] & 0x0F]);
    }
    return out;
}

// Read the entire file at `path` into a buffer. Returns empty on failure.
static bool ReadFileAll(const std::string& path, std::vector<unsigned char>& out) {
    FILE* f = fopen(path.c_str(), "rb");
    if (!f) return false;

    fseek(f, 0, SEEK_END);
    long size = ftell(f);
    if (size < 0) { fclose(f); return false; }
    fseek(f, 0, SEEK_SET);

    out.resize(static_cast<size_t>(size));
    size_t bytesRead = fread(out.data(), 1, out.size(), f);
    fclose(f);
    out.resize(bytesRead);
    return true;
}

// SHA-256 of `data`, returned as a lowercase hex string (CommonCrypto).
static std::string Sha256OfBuffer(const unsigned char* data, size_t len) {
    unsigned char hash[CC_SHA256_DIGEST_LENGTH] = { 0 };
    CC_SHA256(data, static_cast<CC_LONG>(len), hash);
    return ToHex(hash, sizeof(hash));
}

// SHA-256 of the file at `path`, but SKIPS the byte region where
// `skipPattern` appears. This solves the chicken-and-egg problem: the
// embedded EXPECTED_HASH string lives inside the binary, so changing it
// would normally change the hash. By excluding those exact bytes from
// the hash, the computed hash stays stable across the embed step.
//
//   Build 1  (placeholder hash)  -> compute hash-excluding-placeholder = H
//   Set      EXPECTED_HASH = H
//   Build 2  (real hash)         -> compute hash-excluding-real = H  (same!)
//
static std::string Sha256OfFileExcluding(const std::string& path,
                                         const std::string& skipPattern) {
    std::vector<unsigned char> fileData;
    if (!ReadFileAll(path, fileData)) return {};

    // Find the skip region (the 64-char hex string literal in the binary).
    const unsigned char* pat = reinterpret_cast<const unsigned char*>(skipPattern.data());
    size_t patLen = skipPattern.size();
    size_t skipStart = std::string::npos;
    if (patLen > 0 && patLen <= fileData.size()) {
        for (size_t i = 0; i + patLen <= fileData.size(); ++i) {
            if (memcmp(fileData.data() + i, pat, patLen) == 0) {
                skipStart = i;
                break;
            }
        }
    }

    if (skipStart == std::string::npos) {
        // Pattern not found — hash the whole file.
        return Sha256OfBuffer(fileData.data(), fileData.size());
    }

    // Stream the file through CC_SHA256 in two chunks (before + after the skip).
    CC_SHA256_CTX ctx;
    CC_SHA256_Init(&ctx);

    if (skipStart > 0) {
        CC_SHA256_Update(&ctx, fileData.data(), static_cast<CC_LONG>(skipStart));
    }

    size_t after = skipStart + patLen;
    if (after < fileData.size()) {
        CC_SHA256_Update(&ctx, fileData.data() + after,
                         static_cast<CC_LONG>(fileData.size() - after));
    }

    unsigned char hash[CC_SHA256_DIGEST_LENGTH] = { 0 };
    CC_SHA256_Final(hash, &ctx);
    return ToHex(hash, sizeof(hash));
}

// Path of the currently running executable (macOS: via proc_pidpath).
static std::string GetSelfPath() {
    char path[PROC_PIDPATHINFO_MAXSIZE] = { 0 };
    pid_t pid = getpid();
    if (proc_pidpath(pid, path, sizeof(path)) <= 0) return {};
    return std::string(path);
}

// Returns true if the running binary matches the expected SHA-256.
//
// The hash is computed over the binary EXCLUDING the EXPECTED_HASH string
// itself, so embedding the final hash does not change the computed value.
//
// RELEASE WORKFLOW:
//   1. Build with EXPECTED_HASH set to any 64-char hex placeholder.
//   2. Run:  Onyx --print-hash   (prints the exclusion hash, then exits)
//   3. Paste that hash into EXPECTED_HASH.
//   4. Rebuild. The exclusion hash is now stable and matches.
//
// DEV BYPASS: define ONYX_SKIP_INTEGRITY at compile time to disable the
// check entirely during development.
bool VerifyApplicationIntegrity() {
#ifdef ONYX_SKIP_INTEGRITY
    return true;
#endif

    const std::string selfPath = GetSelfPath();
    if (selfPath.empty()) return false;

    const std::string actualHash = Sha256OfFileExcluding(selfPath, EXPECTED_HASH);
    if (actualHash.empty()) return false;

    // Case-insensitive compare.
    return strcasecmp(actualHash.c_str(), EXPECTED_HASH.c_str()) == 0;
}

// Helper: print the exclusion hash of the current binary so you can paste
// it into EXPECTED_HASH.  Usage:  Onyx --print-hash
static void PrintHashAndExit() {
    const std::string selfPath = GetSelfPath();
    const std::string h = Sha256OfFileExcluding(selfPath, EXPECTED_HASH);
    if (h.empty()) {
        std::cerr << "Failed to compute hash.\n";
    } else {
        std::cout << h << "\n";
    }
}

void HandleIntegrityFailure() {
    std::cerr <<
        "Onyx failed its integrity verification.\n\n"
        "This installation may be corrupted or modified. Onyx will not start. "
        "Drag Onyx to the Trash and reinstall it from the official source.\n";

    // Reveal the Applications folder in Finder so the user can trash the app.
    CFURLRef appsURL = CFURLCreateWithFileSystemPath(
        kCFAllocatorDefault,
        CFSTR("/Applications"),
        kCFURLPOSIXPathStyle,
        true);
    if (appsURL) {
        LSOpenCFURLRef(appsURL);
        CFRelease(appsURL);
    }
}

int main(int argc, char* argv[]) {
    // Dev helper:  Onyx --print-hash  -> print the exclusion hash and exit.
    for (int i = 1; i < argc; ++i) {
        if (strcmp(argv[i], "--print-hash") == 0) {
            PrintHashAndExit();
            return 0;
        }
    }

    if (!VerifyApplicationIntegrity()) {
        HandleIntegrityFailure();
        return 1;
    }

    // Normal Onyx startup goes here.
    return 0;
}
