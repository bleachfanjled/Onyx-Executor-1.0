// integrity_check_macOS.cpp — Onyx multi-layer integrity verification (macOS)
//
// Claude's four pillars of code-signing security, all enforced at startup
// and re-enforced at load time:
//
//   1. SHA-256 self-hash (exclusion-based — existing)
//   2. Code signature present and valid (SecStaticCodeCheckValidity)
//   3. Signer identity pinned (Team ID / certificate designation must match
//      the embedded PINNED_TEAM_ID — only Onyx's own Developer ID is accepted)
//   4. Designated requirement satisfied (SecCodeCheckValidity with the
//      signing requirement — verifies the full signing chain)
//
// Load-time re-verification:
//   VerifyFileSignature(path) — re-checks any file's code signature before
//   loading it. An attacker who swaps the file between the startup check
//   and the load call is caught here.
//
// Build (clang):
//   clang++ -std=c++17 integrity_check_macOS.cpp -o Onyx \
//     -framework Security -framework CoreFoundation

#include <CommonCrypto/CommonDigest.h>
#include <Security/Security.h>
#include <CoreFoundation/CoreFoundation.h>
#include <iostream>
#include <string>
#include <vector>
#include <cstring>
#include <unistd.h>
#include <libproc.h>
#include <ApplicationServices/ApplicationServices.h>

// This must match the SHA-256 published on the Onyx download page.
static const std::string EXPECTED_HASH =
    "a3f8c2e1d74b5960fe2318a0c9d47b8e6f1205a3c8d9e7f04b2163a5c8d9e7f0";

// Pinned Team ID — the Apple Developer Program Team ID that signs Onyx.
// Only a binary signed by THIS exact team is accepted. Find your Team ID
// in the Apple Developer portal (it's a 10-character alphanumeric string,
// e.g., "ABCDE12345").
static const std::string PINNED_TEAM_ID = "YOUR_TEAM_ID";

// Pinned designated requirement — a more specific signing requirement that
// pins the exact signing identity. Replace with your actual requirement
// (obtain with:  codesign -d -r- /path/to/Onyx.app).
// Example: "cdhash H\"abcdef1234567890\"" or
//          "identifier \"com.onyx.executor\" and anchor apple generic and ..."
static const std::string PINNED_REQUIREMENT =
    "identifier \"com.onyx.executor\"";

// ============================================================================
// SHA-256 HELPERS (existing, unchanged)
// ============================================================================

static std::string ToHex(const unsigned char* data, size_t len) {
    static const char hex[] = "0123456789abcdef";
    std::string out;
    out.reserve(len * 2);
    for (size_t i = 0; i < len; ++i) {
        out.push_back(hex[(data[i] >> 4) & 0x0F]);
        out.push_back(hex[data[i] & 0x0F]);
    }
    return out;
}

static bool ReadFileAll(const std::string& path, std::vector<unsigned char>& out) {
    FILE* f = fopen(path.c_str(), "rb");
    if (!f) return false;

    fseek(f, 0, SEEK_END);
    long size = ftell(f);
    if (size < 0) { fclose(f); return false; }
    fseek(f, 0, SEEK_SET);

    out.resize(static_cast<size_t>(size));
    size_t bytesRead = fread(out.data(), 1, out.size(), f);
    fclose(f);
    out.resize(bytesRead);
    return true;
}

static std::string Sha256OfBuffer(const unsigned char* data, size_t len) {
    unsigned char hash[CC_SHA256_DIGEST_LENGTH] = { 0 };
    CC_SHA256(data, static_cast<CC_LONG>(len), hash);
    return ToHex(hash, sizeof(hash));
}

static std::string Sha256OfFileExcluding(const std::string& path,
                                         const std::string& skipPattern) {
    std::vector<unsigned char> fileData;
    if (!ReadFileAll(path, fileData)) return {};

    const unsigned char* pat = reinterpret_cast<const unsigned char*>(skipPattern.data());
    size_t patLen = skipPattern.size();
    size_t skipStart = std::string::npos;
    if (patLen > 0 && patLen <= fileData.size()) {
        for (size_t i = 0; i + patLen <= fileData.size(); ++i) {
            if (memcmp(fileData.data() + i, pat, patLen) == 0) {
                skipStart = i;
                break;
            }
        }
    }

    if (skipStart == std::string::npos) {
        return Sha256OfBuffer(fileData.data(), fileData.size());
    }

    CC_SHA256_CTX ctx;
    CC_SHA256_Init(&ctx);

    if (skipStart > 0) {
        CC_SHA256_Update(&ctx, fileData.data(), static_cast<CC_LONG>(skipStart));
    }

    size_t after = skipStart + patLen;
    if (after < fileData.size()) {
        CC_SHA256_Update(&ctx, fileData.data() + after,
                         static_cast<CC_LONG>(fileData.size() - after));
    }

    unsigned char hash[CC_SHA256_DIGEST_LENGTH] = { 0 };
    CC_SHA256_Final(hash, &ctx);
    return ToHex(hash, sizeof(hash));
}

static std::string GetSelfPath() {
    char path[PROC_PIDPATHINFO_MAXSIZE] = { 0 };
    pid_t pid = getpid();
    if (proc_pidpath(pid, path, sizeof(path)) <= 0) return {};
    return std::string(path);
}

// ============================================================================
// LAYER 1: SHA-256 SELF-HASH (existing, unchanged)
// ============================================================================

bool VerifyApplicationIntegrity() {
#ifdef ONYX_SKIP_INTEGRITY
    return true;
#endif

    const std::string selfPath = GetSelfPath();
    if (selfPath.empty()) return false;

    const std::string actualHash = Sha256OfFileExcluding(selfPath, EXPECTED_HASH);
    if (actualHash.empty()) return false;

    return strcasecmp(actualHash.c_str(), EXPECTED_HASH.c_str()) == 0;
}

// ============================================================================
// LAYER 2: CODE SIGNATURE VERIFICATION (SecStaticCodeCheckValidity)
// ============================================================================
//
// Verifies that the binary at `path` has a valid code signature. On macOS,
// this checks:
//   - A signature exists
//   - The signature is structurally valid
//   - The signature has not been tampered with
//   - The binary's content matches what was signed
//
// Returns true only if the signature is present and fully valid.

static std::string CFStringToStd(CFStringRef cfStr) {
    if (!cfStr) return {};
    const char* cStr = CFStringGetCStringPtr(cfStr, kCFStringEncodingUTF8);
    if (cStr) return std::string(cStr);

    CFIndex length = CFStringGetLength(cfStr);
    CFIndex maxSize = CFStringGetMaximumSizeForEncoding(length, kCFStringEncodingUTF8) + 1;
    std::vector<char> buffer(maxSize);
    if (CFStringGetCString(cfStr, buffer.data(), maxSize, kCFStringEncodingUTF8)) {
        return std::string(buffer.data());
    }
    return {};
}

bool VerifyCodeSignature(const std::string& path) {
#ifdef ONYX_SKIP_INTEGRITY
    return true;
#endif

    CFURLRef url = CFURLCreateFromFileSystemRepresentation(
        kCFAllocatorDefault,
        reinterpret_cast<const UInt8*>(path.c_str()),
        static_cast<CFIndex>(path.size()),
        false  // not a directory
    );
    if (!url) return false;

    SecStaticCodeRef staticCode = nullptr;
    OSStatus status = SecStaticCodeCreateWithPath(url, kSecCSDefaultFlags, &staticCode);
    CFRelease(url);

    if (status != errSecSuccess || !staticCode) {
        if (staticCode) CFRelease(staticCode);
        return false;
    }

    // kSecCSCheckAllArchitecture + kSecCSCheckNoOp = thorough validation.
    // kSecCSRequirementInformation ensures the designated requirement is checked.
    SecCSFlags flags = static_cast<SecCSFlags>(
        kSecCSDefaultFlags | kSecCSRequirementInformation
    );

    status = SecStaticCodeCheckValidity(staticCode, flags, nullptr);
    CFRelease(staticCode);

    return status == errSecSuccess;
}

// ============================================================================
// LAYER 3: SIGNER IDENTITY PINNING (Team ID match)
// ============================================================================
//
// Extracts the Team ID from the binary's code signature and compares it
// against PINNED_TEAM_ID. Only a binary signed by Onyx's own Developer
// ID team is accepted. This prevents an attacker from re-signing a
// modified binary with their own Developer ID.

bool VerifySignerIdentity(const std::string& path) {
#ifdef ONYX_SKIP_INTEGRITY
    return true;
#endif

    CFURLRef url = CFURLCreateFromFileSystemRepresentation(
        kCFAllocatorDefault,
        reinterpret_cast<const UInt8*>(path.c_str()),
        static_cast<CFIndex>(path.size()),
        false
    );
    if (!url) return false;

    SecStaticCodeRef staticCode = nullptr;
    OSStatus status = SecStaticCodeCreateWithPath(url, kSecCSDefaultFlags, &staticCode);
    CFRelease(url);

    if (status != errSecSuccess || !staticCode) {
        if (staticCode) CFRelease(staticCode);
        return false;
    }

    // Extract the Team ID from the signing information.
    CFDictionaryRef signingInfo = nullptr;
    status = SecCodeCopySigningInformation(staticCode, kSecCSRequirementInformation, &signingInfo);
    CFRelease(staticCode);

    if (status != errSecSuccess || !signingInfo) {
        if (signingInfo) CFRelease(signingInfo);
        return false;
    }

    std::string teamID;
    CFStringRef teamIDRef = static_cast<CFStringRef>(
        CFDictionaryGetValue(signingInfo, kSecCodeInfoTeamIdentifier)
    );
    if (teamIDRef) {
        teamID = CFStringToStd(teamIDRef);
    }
    CFRelease(signingInfo);

    if (teamID.empty()) return false;

    // Case-sensitive compare — Team IDs are case-sensitive.
    return teamID == PINNED_TEAM_ID;
}

// ============================================================================
// LAYER 4: DESIGNATED REQUIREMENT VERIFICATION
// ============================================================================
//
// Verifies the binary against its designated requirement (DR). The DR is
// a code-signing requirement string embedded in the signature that
// specifies the exact signing identity the binary expects. This is the
// macOS equivalent of full certificate chain verification — it ensures
// the binary was signed by the exact identity it was built to trust.
//
// We also check against our own pinned requirement for extra assurance.

bool VerifyDesignatedRequirement(const std::string& path) {
#ifdef ONYX_SKIP_INTEGRITY
    return true;
#endif

    CFURLRef url = CFURLCreateFromFileSystemRepresentation(
        kCFAllocatorDefault,
        reinterpret_cast<const UInt8*>(path.c_str()),
        static_cast<CFIndex>(path.size()),
        false
    );
    if (!url) return false;

    SecStaticCodeRef staticCode = nullptr;
    OSStatus status = SecStaticCodeCreateWithPath(url, kSecCSDefaultFlags, &staticCode);
    CFRelease(url);

    if (status != errSecSuccess || !staticCode) {
        if (staticCode) CFRelease(staticCode);
        return false;
    }

    // Create a requirement from our pinned string.
    SecRequirementRef requirement = nullptr;
    status = SecRequirementCreateWithString(
        CFStringCreateWithCString(kCFAllocatorDefault,
                                  PINNED_REQUIREMENT.c_str(),
                                  kCFStringEncodingUTF8),
        kSecCSDefaultFlags,
        &requirement
    );

    if (status != errSecSuccess || !requirement) {
        // If we can't create the requirement, fall back to the binary's
        // own designated requirement check.
        if (requirement) CFRelease(requirement);
        status = SecStaticCodeCheckValidity(staticCode,
            static_cast<SecCSFlags>(kSecCSDefaultFlags | kSecCSRequirementInformation),
            nullptr);
        CFRelease(staticCode);
        return status == errSecSuccess;
    }

    // Check the binary against our pinned requirement.
    status = SecStaticCodeCheckValidity(staticCode, kSecCSDefaultFlags, requirement);
    CFRelease(requirement);
    CFRelease(staticCode);

    return status == errSecSuccess;
}

// ============================================================================
// COMBINED VERIFICATION — all four layers must pass
// ============================================================================

struct VerificationResult {
    bool hashOK;
    bool signatureOK;
    bool signerOK;
    bool chainOK;
    bool allOK() const { return hashOK && signatureOK && signerOK && chainOK; }
};

VerificationResult VerifyAll() {
    VerificationResult result = {};

#ifdef ONYX_SKIP_INTEGRITY
    result = { true, true, true, true };
    return result;
#endif

    const std::string selfPath = GetSelfPath();
    if (selfPath.empty()) return result;

    // Layer 1: SHA-256 self-hash.
    const std::string actualHash = Sha256OfFileExcluding(selfPath, EXPECTED_HASH);
    result.hashOK = !actualHash.empty() &&
                    strcasecmp(actualHash.c_str(), EXPECTED_HASH.c_str()) == 0;

    // Layer 2: Code signature present and valid.
    result.signatureOK = VerifyCodeSignature(selfPath);

    // Layer 3: Signer identity (Team ID) pinned.
    result.signerOK = VerifySignerIdentity(selfPath);

    // Layer 4: Designated requirement satisfied.
    result.chainOK = VerifyDesignatedRequirement(selfPath);

    return result;
}

// ============================================================================
// LOAD-TIME RE-VERIFICATION
// ============================================================================
//
// Re-checks a file's code signature AND signer identity before loading it.
// This catches an attacker who swaps the file between the startup check
// and the load call.

bool VerifyFileSignature(const std::string& path) {
#ifdef ONYX_SKIP_INTEGRITY
    return true;
#endif

    if (!VerifyCodeSignature(path)) return false;
    if (!VerifySignerIdentity(path)) return false;
    return true;
}

// ============================================================================
// DEV HELPERS + FAILURE HANDLING
// ============================================================================

static void PrintHashAndExit() {
    const std::string selfPath = GetSelfPath();
    const std::string h = Sha256OfFileExcluding(selfPath, EXPECTED_HASH);
    if (h.empty()) {
        std::cerr << "Failed to compute hash.\n";
    } else {
        std::cout << h << "\n";
    }
}

void HandleIntegrityFailure() {
    std::cerr <<
        "Onyx failed its integrity verification.\n\n"
        "This installation may be corrupted, modified, or signed by an "
        "untrusted identity. Onyx will not start. Drag Onyx to the Trash "
        "and reinstall it from the official source.\n";

    CFURLRef appsURL = CFURLCreateWithFileSystemPath(
        kCFAllocatorDefault,
        CFSTR("/Applications"),
        kCFURLPOSIXPathStyle,
        true);
    if (appsURL) {
        LSOpenCFURLRef(appsURL);
        CFRelease(appsURL);
    }
}

#ifndef ONYX_ADDON_BUILD
int main(int argc, char* argv[]) {
    for (int i = 1; i < argc; ++i) {
        if (strcmp(argv[i], "--print-hash") == 0) {
            PrintHashAndExit();
            return 0;
        }
    }

    // Run all four verification layers.
    VerificationResult result = VerifyAll();
    if (!result.allOK()) {
        HandleIntegrityFailure();
        return 1;
    }

    // Normal Onyx startup goes here.
    return 0;
}
#endif  // ONYX_ADDON_BUILD
