// =============================================================================
// ONYX EXECUTOR — main.js
// Electron main process. Creates the app window and wires up IPC handlers.
// =============================================================================

const { app, BrowserWindow, ipcMain, dialog } = require("electron");
const path = require("path");
const { execSync } = require("child_process");

// Set the AppUserModelID so Windows groups the window correctly on the
// taskbar and the pinned shortcut resolves to the installed Onyx.exe
// (not to electron.exe or a loose renderer.js file).
if (process.platform === "win32") {
  app.setAppUserModelId("com.onyx.executor");
}

let mainWindow = null;

// ---- C++ backend (native addon) ----
// Build it first:  npm run rebuild   (needs node-gyp + Visual Studio Build Tools)
// The app still runs without it — the renderer falls back to local JS behavior —
// but attach/execute/load/save only hit the C++ shell once onyx.node is built.
let onyxNative = null;
try {
  onyxNative = require("./build/Release/onyx.node");
  onyxNative.init();
} catch (err) {
  console.error("[Onyx] Native backend not loaded. Run `npm run rebuild` to build it.\n", err);
}

// Catch uncaught exceptions in the main process so we can log them
// instead of silently crashing with a generic "JavaScript error" dialog.
process.on("uncaughtException", (err) => {
  const msg = err && err.stack ? err.stack : String(err);
  dialog.showErrorBox(
    "Onyx — Main Process Error",
    "An unexpected error occurred:\n\n" + msg
  );
});

function createWindow() {
  // On macOS use a hidden title bar so the native traffic-light controls
  // (close / minimize / maximize) remain available on the left inset.
  // On Windows keep a frameless window with custom controls in the renderer.
  const isMac = process.platform === "darwin";
  mainWindow = new BrowserWindow({
    width: 900,
    height: 620,
    resizable: true,
    frame: !isMac,
    titleBarStyle: isMac ? "hiddenInset" : "default",
    trafficLightPosition: isMac ? { x: 12, y: 12 } : undefined,
    backgroundColor: "#050505",
    title: "Onyx",
    webPreferences: {
      preload: path.join(__dirname, "preload.js"),
      contextIsolation: true,
      nodeIntegration: false,
    },
  });

  mainWindow.loadFile(path.join(__dirname, "index.html"));

  mainWindow.on("closed", () => { mainWindow = null; });

  // Surface renderer/preload load failures to the user instead of a blank window.
  mainWindow.webContents.on("did-fail-load", (_e, errorCode, errorDescription) => {
    dialog.showErrorBox(
      "Onyx — Load Error",
      "Failed to load the application window.\n\nCode " +
        errorCode + ": " + (errorDescription || "Unknown error")
    );
  });
}

// IPC handlers for window controls
ipcMain.on("window-minimize", () => {
  if (mainWindow) mainWindow.minimize();
});

ipcMain.on("window-close", () => {
  if (mainWindow) mainWindow.close();
});

ipcMain.on("window-maximize", () => {
  if (!mainWindow) return;
  if (mainWindow.isMaximized()) mainWindow.unmaximize();
  else mainWindow.maximize();
});

ipcMain.on("window-fullscreen", () => {
  if (!mainWindow) return;
  mainWindow.setFullScreen(!mainWindow.isFullScreen());
});

// ---- Kill Roblox ----
// Terminate every running process whose name contains "Roblox" (launcher
// "Roblox", game client "RobloxPlayerBeta", helpers). This is a Windows
// taskkill via PowerShell — it lists processes by name and force-terminates
// them; it does NOT read Roblox memory or inject into it.
ipcMain.handle("onyx:kill-roblox", () => {
  try {
    execSync(
      'powershell -NoProfile -Command "Get-Process | Where-Object { $_.ProcessName -like \'*Roblox*\' } | Stop-Process -Force"',
      { stdio: "ignore" }
    );
  } catch (err) {
    // PowerShell unavailable or nothing matched — non-fatal.
  }
  return {
    lines: [{ text: "[ ONYX ] Roblox Processes Terminated", color: "#FFFFFF" }],
    status: { text: "Idle", active: false },
    attached: false,
  };
});

ipcMain.handle("onyx:execute", (_e, script) =>
  onyxNative ? onyxNative.execute(String(script || "")) : null
);

ipcMain.handle("onyx:load-file", async () => {
  if (!onyxNative) return null;
  const r = await dialog.showOpenDialog(mainWindow, {
    title: "Load Script",
    filters: [{ name: "Lua / Text", extensions: ["lua", "txt", "luac"] }],
    properties: ["openFile"],
  });
  if (r.canceled || !r.filePaths.length) return null;
  return onyxNative.loadFile(r.filePaths[0]);
});

ipcMain.handle("onyx:save-file", async (_e, content) => {
  if (!onyxNative) return null;
  const r = await dialog.showSaveDialog(mainWindow, {
    title: "Save Script",
    defaultPath: "onyx_script.lua",
    filters: [{ name: "Lua / Text", extensions: ["lua", "txt", "luac"] }],
  });
  if (r.canceled || !r.filePath) return null;
  return onyxNative.saveFile(r.filePath, String(content || ""));
});

app.whenReady().then(createWindow).catch((err) => {
  dialog.showErrorBox(
    "Onyx — Startup Error",
    "Failed to create the application window:\n\n" +
      (err && err.stack ? err.stack : String(err))
  );
});

app.on("window-all-closed", () => {
  if (process.platform !== "darwin") app.quit();
});

app.on("activate", () => {
  if (BrowserWindow.getAllWindows().length === 0) createWindow();
});
