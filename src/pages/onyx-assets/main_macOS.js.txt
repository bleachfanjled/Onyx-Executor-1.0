// =============================================================================
// ONYX EXECUTOR — main_macOS.js
// Electron main process for macOS builds.
// Uses a hidden title bar with native traffic-light controls and drops all
// Windows-only setup (AppUserModelID, NSIS, etc.). The C++ native addon is
// optional — on macOS it typically won't be built, so the renderer falls back
// to local JS behavior. attach/execute/load/save only hit the C++ shell once
// onyx.node is present.
// =============================================================================

const { app, BrowserWindow, ipcMain, dialog } = require("electron");
const path = require("path");
const { execSync } = require("child_process");

let mainWindow = null;

// ---- C++ backend (native addon) ----
// Optional on macOS. If you build it (`npm run rebuild`), it loads; otherwise
// the app still runs and the renderer handles everything in JS.
let onyxNative = null;
try {
  onyxNative = require("./build/Release/onyx.node");
  onyxNative.init();
} catch (err) {
  console.error("[Onyx/macOS] Native backend not loaded. Run `npm run rebuild` to build it.\n", err);
}

// Log uncaught exceptions instead of silently crashing.
process.on("uncaughtException", (err) => {
  const msg = err && err.stack ? err.stack : String(err);
  dialog.showErrorBox(
    "Onyx — Main Process Error",
    "An unexpected error occurred:\n\n" + msg
  );
});

function createWindow() {
  // macOS: hidden title bar so the native traffic-light controls
  // (close / minimize / maximize) sit on the left inset.
  mainWindow = new BrowserWindow({
    width: 900,
    height: 620,
    resizable: true,
    frame: false,
    titleBarStyle: "hiddenInset",
    trafficLightPosition: { x: 12, y: 12 },
    backgroundColor: "#050505",
    title: "Onyx",
    webPreferences: {
      preload: path.join(__dirname, "preload.js"),
      contextIsolation: true,
      nodeIntegration: false,
    },
  });

  mainWindow.loadFile(path.join(__dirname, "index.html"));

  mainWindow.on("closed", () => { mainWindow = null; });

  // Surface load failures instead of a blank window.
  mainWindow.webContents.on("did-fail-load", (_e, errorCode, errorDescription) => {
    dialog.showErrorBox(
      "Onyx — Load Error",
      "Failed to load the application window.\n\nCode " +
        errorCode + ": " + (errorDescription || "Unknown error")
    );
  });
}

// ---- Window control IPC ----
// On macOS the traffic lights handle close/minimize/maximize natively, but
// the renderer's custom controls (if shown) route through these handlers too.
ipcMain.on("window-minimize", () => {
  if (mainWindow) mainWindow.minimize();
});

ipcMain.on("window-close", () => {
  if (mainWindow) mainWindow.close();
});

ipcMain.on("window-maximize", () => {
  if (!mainWindow) return;
  if (mainWindow.isMaximized()) mainWindow.unmaximize();
  else mainWindow.maximize();
});

// ---- C++ backend IPC (renderer -> main -> native addon) ----
// On macOS there is no real process attach (no injection). The "Attach" click
// just flips the UI into the attached state; the "Kill Roblox" click actually
// terminates the Roblox process(es) — equivalent to macOS Force Quit. This runs
// in the main process via Node, so it works even without the native addon built.
ipcMain.handle("onyx:attach-toggle", (_e, currentlyAttached) => {
  if (currentlyAttached) {
    // Kill Roblox: terminate the Roblox player + app processes (graceful SIGTERM).
    try {
      execSync(
        'killall RobloxPlayer 2>/dev/null; killall Roblox 2>/dev/null; pkill -f "Roblox.app/Contents/MacOS" 2>/dev/null; true',
        { stdio: "ignore" }
      );
    } catch (err) {
      // pkill/killall return non-zero when nothing matches — that's fine.
    }
    return {
      lines: [{ text: "[ ONYX ] Roblox Processes Terminated", color: "#FFFFFF" }],
      status: { text: "Idle", active: false },
      attached: false,
    };
  }
  // Attach: no real attach on macOS — just report the attached state.
  return {
    lines: [{ text: "[ ONYX ] Roblox Client Attached", color: "#FFFFFF" }],
    status: { text: "Attached", active: true },
    attached: true,
  };
});

ipcMain.handle("onyx:execute", (_e, script) =>
  onyxNative ? onyxNative.execute(String(script || "")) : null
);

ipcMain.handle("onyx:load-file", async () => {
  if (!onyxNative) return null;
  const r = await dialog.showOpenDialog(mainWindow, {
    title: "Load Script",
    filters: [{ name: "Lua / Text", extensions: ["lua", "txt", "luac"] }],
    properties: ["openFile"],
  });
  if (r.canceled || !r.filePaths.length) return null;
  return onyxNative.loadFile(r.filePaths[0]);
});

ipcMain.handle("onyx:save-file", async (_e, content) => {
  if (!onyxNative) return null;
  const r = await dialog.showSaveDialog(mainWindow, {
    title: "Save Script",
    defaultPath: "onyx_script.lua",
    filters: [{ name: "Lua / Text", extensions: ["lua", "txt", "luac"] }],
  });
  if (r.canceled || !r.filePath) return null;
  return onyxNative.saveFile(r.filePath, String(content || ""));
});

app.whenReady().then(createWindow).catch((err) => {
  dialog.showErrorBox(
    "Onyx — Startup Error",
    "Failed to create the application window:\n\n" +
      (err && err.stack ? err.stack : String(err))
  );
});

// macOS keeps apps alive in the dock after all windows close.
app.on("window-all-closed", () => {
  if (process.platform !== "darwin") app.quit();
});

app.on("activate", () => {
  if (BrowserWindow.getAllWindows().length === 0) createWindow();
});
