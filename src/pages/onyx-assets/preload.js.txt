// =============================================================================
// ONYX EXECUTOR — preload.js
// Bridges IPC to the renderer:
//   window.electronAPI.minimize() / .close()   -> window controls
//   window.onyx.attachToggle() / execute() / loadFile() / saveFile()
//     -> C++ backend (onyx.node) via main process IPC
// =============================================================================

const { contextBridge, ipcRenderer } = require("electron");

contextBridge.exposeInMainWorld("electronAPI", {
  platform: process.platform,
  minimize: () => ipcRenderer.send("window-minimize"),
  maximize: () => ipcRenderer.send("window-maximize"),
  fullscreen: () => ipcRenderer.send("window-fullscreen"),
  close: () => ipcRenderer.send("window-close"),
  onClosing: (callback) => ipcRenderer.on("app-closing", callback),
  respondClose: (shouldClose) => ipcRenderer.send("onyx:close-response", shouldClose),
});

// C++ backend bridge — renderer calls these; main process forwards to onyx.node
contextBridge.exposeInMainWorld("onyx", {
  killRoblox: () => ipcRenderer.invoke("onyx:kill-roblox"),
  execute: (script) => ipcRenderer.invoke("onyx:execute", script),
  loadFile: () => ipcRenderer.invoke("onyx:load-file"),
  saveFile: (content) => ipcRenderer.invoke("onyx:save-file", content),
  verifyIntegrity: () => ipcRenderer.invoke("onyx:verify-integrity"),
  verifyFile: (filePath) => ipcRenderer.invoke("onyx:verify-file", filePath),
});
