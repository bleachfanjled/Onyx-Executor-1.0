// Structure passed to our shellcode loader to resolve environment dynamically
struct MAPPING_DATA {
    PIMAGE_NT_HEADERS NtHeaders;
    PBYTE BaseAddress;

    typedef HMODULE(WINAPI* fnLoadLibraryA)(LPCSTR);
    typedef FARPROC(WINAPI* fnGetProcAddress)(HMODULE, LPCSTR);

    fnLoadLibraryA pLoadLibraryA;
    fnGetProcAddress pProcAddress;
};

static DWORD WINAPI ShellcodeLoader(MAPPING_DATA* pData) {
    if (!pData) return 0;

    PIMAGE_NT_HEADERS pOldNtHeader = pData->NtHeaders;
    PBYTE pBase = pData->BaseAddress;

    auto* pLocationDelta = (PBYTE)(pBase - pOldNtHeader->OptionalHeader.ImageBase);
    if (pLocationDelta) {
        if (pOldNtHeader->OptionalHeader.DataDirectory[IMAGE_DIRECTORY_ENTRY_BASERELOC].Size) {
            auto* pRelocData = (PIMAGE_BASE_RELOCATION)(pBase + pOldNtHeader->OptionalHeader.DataDirectory[IMAGE_DIRECTORY_ENTRY_BASERELOC].VirtualAddress);
            while (pRelocData->VirtualAddress) {
                UINT AmountOfEntries = (pRelocData->SizeOfBlock - sizeof(IMAGE_BASE_RELOCATION)) / sizeof(WORD);
                WORD* pRelativeInfo = (WORD*)(pRelocData + 1);

                for (UINT i = 0; i < AmountOfEntries; ++i) {
                    if ((pRelativeInfo[i] >> 12) == IMAGE_REL_BASED_DIR64) {
                        ULONG_PTR* pPatch = (ULONG_PTR*)(pBase + pRelocData->VirtualAddress + (pRelativeInfo[i] & 0xFFF));
                        *pPatch += (ULONG_PTR)pLocationDelta;
                    }
                }
                pRelocData = (PIMAGE_BASE_RELOCATION)((PBYTE)pRelocData + pRelocData->SizeOfBlock);
            }
        }
    }

    if (pOldNtHeader->OptionalHeader.DataDirectory[IMAGE_DIRECTORY_ENTRY_IMPORT].Size) {
        auto* pImportDescr = (PIMAGE_IMPORT_DESCRIPTOR)(pBase + pOldNtHeader->OptionalHeader.DataDirectory[IMAGE_DIRECTORY_ENTRY_IMPORT].VirtualAddress);
        while (pImportDescr->Name) {
            char* szMod = (char*)(pBase + pImportDescr->Name);
            HMODULE hMod = pData->pLoadLibraryA(szMod);

            ULONG_PTR* pThunkRef = (ULONG_PTR*)(pBase + pImportDescr->OriginalFirstThunk);
            ULONG_PTR* pFuncRef = (ULONG_PTR*)(pBase + pImportDescr->FirstThunk);

            if (!pThunkRef) pThunkRef = pFuncRef;

            for (; *pThunkRef; ++pThunkRef, ++pFuncRef) {
                if (IMAGE_SNAP_BY_ORDINAL(*pThunkRef)) {
                    *pFuncRef = (ULONG_PTR)pData->pProcAddress(hMod, (LPCSTR)(*pThunkRef & 0xFFFF));
                } else {
                    auto* pImportByName = (PIMAGE_IMPORT_BY_NAME)(pBase + (*pThunkRef));
                    *pFuncRef = (ULONG_PTR)pData->pProcAddress(hMod, pImportByName->Name);
                }
            }
            pImportDescr++;
        }
    }

    if (pOldNtHeader->OptionalHeader.DataDirectory[IMAGE_DIRECTORY_ENTRY_TLS].Size) {
        auto* pTLSDir = (PIMAGE_TLS_DIRECTORY)(pBase + pOldNtHeader->OptionalHeader.DataDirectory[IMAGE_DIRECTORY_ENTRY_TLS].VirtualAddress);
        auto* pCallback = (PIMAGE_TLS_CALLBACK*)pTLSDir->AddressOfCallBacks;
        while (pCallback && *pCallback) {
            (*pCallback)(pBase, DLL_PROCESS_ATTACH, nullptr);
            pCallback++;
        }
    }

    typedef BOOL(WINAPI* fnDllMain)(HINSTANCE, DWORD, LPVOID);
    auto pDllMain = (fnDllMain)(pBase + pOldNtHeader->OptionalHeader.AddressOfEntryPoint);
    return pDllMain((HINSTANCE)pBase, DLL_PROCESS_ATTACH, nullptr);
}

static void ShellcodeLoaderEnd() {}

DWORD GetTargetThreadId(DWORD processId) {
    HANDLE hSnapshot = CreateToolhelp32Snapshot(TH32CS_SNAPTHREAD, 0);
    if (hSnapshot == INVALID_HANDLE_VALUE) return 0;

    THREADENTRY32 te;
    te.dwSize = sizeof(te);
    if (Thread32First(hSnapshot, &te)) {
        do {
            if (te.th32OwnerProcessID == processId) {
                CloseHandle(hSnapshot);
                return te.th32ThreadID;
            }
        } while (Thread32Next(hSnapshot, &te));
    }
    CloseHandle(hSnapshot);
    return 0;
}

bool ManualMapAndHijack(HANDLE hProc, DWORD targetThreadId, BYTE* pSrcData) {
    if (!hProc || !pSrcData || !targetThreadId) return false;

    IMAGE_DOS_HEADER* pDosHeader = (IMAGE_DOS_HEADER*)pSrcData;
    IMAGE_NT_HEADERS* pOldNtHeader = (IMAGE_NT_HEADERS*)(pSrcData + pDosHeader->e_lfanew);

    BYTE* pTargetBase = (BYTE*)VirtualAllocEx(hProc, nullptr, pOldNtHeader->OptionalHeader.SizeOfImage, MEM_COMMIT | MEM_RESERVE, PAGE_EXECUTE_READWRITE);
    if (!pTargetBase) return false;

    SIZE_T bytesWritten = 0;
    if (!WriteProcessMemory(hProc, pTargetBase, pSrcData, pOldNtHeader->OptionalHeader.SizeOfHeaders, &bytesWritten)) return false;

    IMAGE_SECTION_HEADER* pSectionHeader = IMAGE_FIRST_SECTION(pOldNtHeader);
    for (UINT i = 0; i < pOldNtHeader->FileHeader.NumberOfSections; ++i) {
        SIZE_T sectionBytes = 0;
        if (!WriteProcessMemory(hProc, pTargetBase + pSectionHeader[i].VirtualAddress, pSrcData + pSectionHeader[i].PointerToRawData, pSectionHeader[i].SizeOfRawData, &sectionBytes)) {
            return false;
        }
    }

    MAPPING_DATA data = { 0 };
    data.NtHeaders = (PIMAGE_NT_HEADERS)(pTargetBase + pDosHeader->e_lfanew);
    data.BaseAddress = pTargetBase;
    data.pLoadLibraryA = LoadLibraryA;
    data.pProcAddress = (MAPPING_DATA::fnGetProcAddress)GetProcAddress;

    DWORD shellcodeSize = (DWORD)((LPBYTE)ShellcodeLoaderEnd - (LPBYTE)ShellcodeLoader);
    BYTE* pMappingDataRemote = (BYTE*)VirtualAllocEx(hProc, nullptr, sizeof(MAPPING_DATA) + shellcodeSize, MEM_COMMIT | MEM_RESERVE, PAGE_EXECUTE_READWRITE);
    if (!pMappingDataRemote) return false;

    BYTE* pShellcodeRemote = pMappingDataRemote + sizeof(MAPPING_DATA);
    if (!WriteProcessMemory(hProc, pMappingDataRemote, &data, sizeof(MAPPING_DATA), nullptr)) return false;
    if (!WriteProcessMemory(hProc, pShellcodeRemote, (LPCVOID)ShellcodeLoader, shellcodeSize, nullptr)) return false;

    HANDLE hThread = OpenThread(THREAD_GET_CONTEXT | THREAD_SET_CONTEXT | THREAD_SUSPEND_RESUME, FALSE, targetThreadId);
    if (!hThread) return false;

    SuspendThread(hThread);

    CONTEXT ctx;
    ctx.ContextFlags = CONTEXT_CONTROL;
    if (!GetThreadContext(hThread, &ctx)) {
        CloseHandle(hThread);
        return false;
    }

#if defined(_WIN64)
    ctx.Rip = (DWORD64)pShellcodeRemote;
    ctx.Rcx = (DWORD64)pMappingDataRemote;
#else
    ctx.Eip = (DWORD)pShellcodeRemote;
#endif

    if (!SetThreadContext(hThread, &ctx)) {
        CloseHandle(hThread);
        return false;
    }

    ResumeThread(hThread);
    CloseHandle(hThread);

    return true;
}
`;

function appendConsoleLine(text, color = "#959595") {
  const line = document.createElement("p");
  line.className = "text-[11px]";
  line.style.color = color;
  line.textContent = text;
  consoleEl.appendChild(line);
  consoleEl.scrollTop = consoleEl.scrollHeight;
}

function log(msg, color = "#959595") {
  appendConsoleLine("[ ONYX ] " + msg, color);
}

// Apply a result returned from the C++ backend: append its console lines,
// update status, and sync the attach button.
function applyBackendResult(res) {
  if (!res) return;
  if (Array.isArray(res.lines)) {
    res.lines.forEach((l) => appendConsoleLine(l.text, l.color || "#959595"));
  }
  if (res.status) setStatus(res.status.text, res.status.active);
  if (typeof res.attached === "boolean") {
    clientAttached = res.attached;
    syncAttachButton();
  }
}

function showNotification(msg) {
  toast.textContent = msg;
  toast.classList.remove("visible");
  void toast.offsetWidth;
  toast.classList.add("visible");

  clearTimeout(showNotification.timeoutId);
  showNotification.timeoutId = setTimeout(() => {
    toast.classList.remove("visible");
  }, 1800);
}

function setStatus(text, active = false) {
  statusText.textContent = text;
  statusDot.style.backgroundColor = active ? "#FFFFFF" : "#3A3A3A";
}

function syncAttachButton() {
  if (!attachButton) return;
  attachButton.textContent = clientAttached ? "Detach" : "Attach";
}

function getActiveTab() {
  return state.tabs.find((tab) => tab.id === selectedTabId) || state.tabs[0];
}

function saveActiveTabContent() {
  const activeTab = getActiveTab();
  if (!activeTab) return;
  activeTab.content = editor.value;
}

function renameTab(tabId) {
  const tab = state.tabs.find((item) => item.id === tabId);
  if (!tab) return;

  saveActiveTabContent();
  selectedTabId = tabId;
  editor.value = tab.content;
  renderTabs();

  const tabWrap = tabsEl.querySelector(`[data-tab-id="${tabId}"]`);
  if (!tabWrap) return;

  const currentLabel = tabWrap.querySelector(".tab-label");
  if (!currentLabel) return;

  const input = document.createElement("input");
  input.type = "text";
  input.value = tab.name;
  input.className = "tab-label w-[90px] bg-transparent border border-white px-2 text-[10px] uppercase tracking-widest text-white outline-none";
  input.setAttribute("maxlength", "24");

  const commitRename = () => {
    const nextName = input.value.trim();
    if (nextName) tab.name = nextName;
    renderTabs();
  };

  input.addEventListener("keydown", (event) => {
    if (event.key === "Enter") commitRename();
    if (event.key === "Escape") renderTabs();
  });
  input.addEventListener("blur", commitRename);

  currentLabel.replaceWith(input);
  input.focus();
  input.select();
}

function renderTabs() {
  tabsEl.innerHTML = "";

  state.tabs.forEach((tab) => {
    const tabWrap = document.createElement("div");
    tabWrap.dataset.tabId = tab.id;
    tabWrap.className = "flex items-center h-7 border transition-all " + (
      selectedTabId === tab.id
        ? "border-white bg-[#0F0F0F] text-white"
        : "border-[#1A1A1A] text-[#959595] hover:border-[#3A3A3A] hover:text-white"
    );

    const tabBtn = document.createElement("button");
    tabBtn.type = "button";
    tabBtn.dataset.id = tab.id;
    tabBtn.className = "tab-label px-3 text-[10px] uppercase tracking-widest whitespace-nowrap";
    tabBtn.textContent = tab.name;
    tabBtn.addEventListener("click", (event) => {
      event.preventDefault();
      event.stopPropagation();
      saveActiveTabContent();
      selectedTabId = tab.id;
      editor.value = tab.content;
      renderTabs();
    });
    tabBtn.addEventListener("dblclick", (event) => {
      event.preventDefault();
      event.stopPropagation();
      renameTab(tab.id);
    });
    tabBtn.addEventListener("contextmenu", (event) => {
      event.preventDefault();
      event.stopPropagation();
      contextTabId = tab.id;
      contextMenu.classList.remove("hidden");
      contextMenu.style.left = `${event.clientX}px`;
      contextMenu.style.top = `${event.clientY}px`;
    });

    const closeBtn = document.createElement("button");
    closeBtn.type = "button";
    closeBtn.textContent = "×";
    closeBtn.className = "w-6 h-full text-[12px] border-l border-[#1A1A1A] text-[#959595] hover:text-white hover:bg-[#1A1A1A]";
    closeBtn.addEventListener("click", (event) => {
      event.preventDefault();
      event.stopPropagation();
      closeTab(tab.id);
    });

    tabWrap.appendChild(tabBtn);
    tabWrap.appendChild(closeBtn);
    tabsEl.appendChild(tabWrap);
  });

  const addButton = document.createElement("button");
  addButton.type = "button";
  addButton.textContent = "+";
  addButton.className = "h-7 w-7 border border-[#1A1A1A] text-[#959595] hover:text-white hover:border-[#3A3A3A] transition-all";
  addButton.addEventListener("click", () => {
    saveActiveTabContent();
    const newTab = { id: crypto.randomUUID(), name: "New Script", content: "" };
    state.tabs.push(newTab);
    selectedTabId = newTab.id;
    editor.value = newTab.content;
    renderTabs();
  });
  tabsEl.appendChild(addButton);
}

function closeTab(tabId) {
  if (state.tabs.length === 1) {
    const tab = getActiveTab();
    tab.content = "";
    tab.name = "New Script";
    editor.value = "";
    renderTabs();
    return;
  }

  state.tabs = state.tabs.filter((tab) => tab.id !== tabId);
  selectedTabId = state.tabs[0].id;
  editor.value = getActiveTab().content;
  renderTabs();
}

function handleAttach() {
  clientAttached = !clientAttached;
  syncAttachButton();
  setStatus(clientAttached ? "Attached" : "Idle", clientAttached);
  log(clientAttached ? "Attached to Roblox." : "Detached from Roblox.", "#FFFFFF");
  showNotification(clientAttached ? "Attached" : "Detached");
}

async function handleKillRoblox() {
  const res = window.onyx ? await window.onyx.killRoblox() : null;
  if (res) {
    applyBackendResult(res);
    showNotification("Roblox Processes Terminated");
    return;
  }
  // Fallback (native backend not built yet): log only.
  log("Roblox Processes Terminated", "#FFFFFF");
  setStatus("Idle", false);
  showNotification("Roblox Processes Terminated");
}

async function handleExecute() {
  const script = editor.value.trim();
  if (!script) {
    log("No script to execute.", "#3A3A3A");
    showNotification("No script to execute.");
    return;
  }

  if (window.onyx) {
    const res = await window.onyx.execute(script);
    if (res) {
      applyBackendResult(res);
      return;
    }
  }

  // Fallback (native backend not built): local checks only.
  if (!clientAttached) {
    log("Attach to Roblox before executing.", "#FFFFFF");
    showNotification("Attach to Roblox before executing.");
    return;
  }
  log("Execution not implemented.", "#3A3A3A");
  showNotification("Execution not implemented.");
}

function handleCopy() {
  const script = editor.value;
  if (!script) { log("Nothing to copy.", "#3A3A3A"); showNotification("Nothing to copy."); return; }
  navigator.clipboard.writeText(script).then(() => { log("Script copied to clipboard."); showNotification("Script copied."); }).catch(() => { log("Clipboard access denied.", "#3A3A3A"); showNotification("Clipboard access denied."); });
}

async function handlePaste() {
  try {
    const text = await navigator.clipboard.readText();
    editor.value += text;
    saveActiveTabContent();
    log("Pasted from clipboard.");
    showNotification("Pasted from clipboard.");
  } catch {
    log("Clipboard access denied.", "#3A3A3A");
    showNotification("Clipboard access denied.");
  }
}

async function handleDownload() {
  const script = editor.value;
  if (!script) { log("Nothing to save.", "#3A3A3A"); return; }
  if (window.onyx) {
    const res = await window.onyx.saveFile(script);
    if (res) {
      applyBackendResult(res);
      if (res.ok) showNotification("Script saved.");
      return;
    }
  }
  // Fallback: browser download
  const blob = new Blob([script], { type: "text/plain" });
  const url = URL.createObjectURL(blob);
  const a = document.createElement("a");
  a.href = url;
  a.download = "onyx_script.lua";
  a.click();
  URL.revokeObjectURL(url);
  log("Script saved as onyx_script.lua.");
  showNotification("Script saved.");
}

async function handleLoad() {
  if (window.onyx) {
    const res = await window.onyx.loadFile();
    if (res) {
      applyBackendResult(res);
      if (res.ok) {
        saveActiveTabContent();
        const activeTab = getActiveTab();
        if (activeTab) {
          activeTab.content = res.content || "";
          activeTab.name = res.name || "Loaded Script";
          editor.value = activeTab.content;
          renderTabs();
          showNotification(`Loaded ${res.name || "script"}`);
        }
      }
      return;
    }
  }
  // Fallback: browser file picker
  const input = document.createElement("input");
  input.type = "file";
  input.accept = ".lua,.txt,.luac";
  input.addEventListener("change", (event) => {
    const file = event.target.files && event.target.files[0];
    if (!file) return;
    const reader = new FileReader();
    reader.onload = (e) => {
      saveActiveTabContent();
      const activeTab = getActiveTab();
      if (!activeTab) return;
      activeTab.content = String(e.target.result || "");
      activeTab.name = file.name.replace(/\.[^.]+$/, "") || "Loaded Script";
      editor.value = activeTab.content;
      renderTabs();
      log(`Loaded ${file.name}.`);
      showNotification(`Loaded ${file.name}`);
    };
    reader.onerror = () => log("Failed to read file.", "#3A3A3A");
    reader.readAsText(file);
  });
  input.click();
}

function handleClear() {
  editor.value = "";
  saveActiveTabContent();
  log("Editor cleared.");
  showNotification("Editor cleared.");
}

function initOnboarding() {
  onboarding.style.opacity = "1";
  verifyScreen.style.opacity = "1";
  startVerification();
}

function startVerification() {
  const steps = [
    { text: "Initializing verification...", progress: 0.25 },
    { text: "Computing SHA-256 hash...", progress: 0.50 },
    { text: "Checking digital signature...", progress: 0.75 },
    { text: "Integrity verified.", progress: 1.0 },
  ];
  let i = 0;
  function next() {
    if (i >= steps.length) {
      verifyCheck.style.animation = "checkmark 0.4s ease-out forwards";
      setTimeout(showWelcome, 800);
      return;
    }
    verifyStatus.textContent = steps[i].text;
    verifyProgress.style.strokeDashoffset = 226 * (1 - steps[i].progress);
    i++;
    setTimeout(next, 550);
  }
  next();
}

function showWelcome() {
  verifyScreen.style.opacity = "0";

    welcomeScreen.style.display = "flex";
    requestAnimationFrame(() => { welcomeScreen.style.opacity = "1"; });
  }, 400);
}

function showTutorialStep(step) {
  currentStep = step;
  if (step >= tutorialSteps.length) { closeOnboarding(); return; }
  if (spotlightEl) { spotlightEl.classList.remove("onyx-spotlight"); spotlightEl = null; }
  const stepData = tutorialSteps[step];
  const target = document.querySelector(stepData.target);
  if (!target) { showTutorialStep(step + 1); return; }
  spotlightEl = target;
  target.classList.add("onyx-spotlight");
  const rect = target.getBoundingClientRect();
  tutorialTooltip.style.display = "block";
  tutorialStepNum.textContent = `Step ${step + 1} / ${tutorialSteps.length}`;
  tutorialTitle.textContent = stepData.title;
  tutorialDesc.textContent = stepData.desc;
  renderDots();
  const tooltipW = 320;
  let left = rect.left + rect.width / 2 - tooltipW / 2;
  let top = rect.bottom + 16;
  left = Math.max(16, Math.min(left, window.innerWidth - tooltipW - 16));
  if (top + 220 > window.innerHeight) { top = Math.max(16, rect.top - 220); }
  tutorialTooltip.style.left = `${left}px`;
  tutorialTooltip.style.top = `${top}px`;
  requestAnimationFrame(() => { tutorialTooltip.style.opacity = "1"; });
  tutorialBack.style.opacity = step === 0 ? "0.3" : "1";
  tutorialBack.style.pointerEvents = step === 0 ? "none" : "auto";
  tutorialNext.textContent = step === tutorialSteps.length - 1 ? "Finish ✓" : "Next →";
}

function closeOnboarding() {
  if (spotlightEl) { spotlightEl.classList.remove("onyx-spotlight"); spotlightEl = null; }
  tutorialTooltip.style.opacity = "0";
  onboarding.style.opacity = "0";
  setTimeout(() => { onboarding.style.display = "none"; }, 500);
}
